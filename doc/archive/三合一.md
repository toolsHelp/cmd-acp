# 一、ACP Trace Shim 目标

不要直接修改 Paseo。

链路：

```
                 Paseo
                   |
                   |
              spawn ACP Agent
                   |
                   |
              codebuddy-code
                   |
                   |
              ACP stdio JSON-RPC


改成：

                 Paseo
                   |
                   |
             codebuddy-code.cmd
                   |
                   |
              ACP Trace Shim
                   |
        +----------+----------+
        |                     |
      log                  forward
        |                     |
 trace.jsonl            real ACP agent


```

Shim 做三件事：

1. 截获 stdin JSON-RPC
2. 记录 ACP 原始消息
3. 原样转发 stdout/stderr

不要改变协议。

否则 Paseo 会认为 ACP Agent 异常。

# 八、完整 probe.js

```javascript
const fs=require("fs");
const {spawn}=require("child_process");


const logFile=
"C:\\acp-trace-shim\\logs\\trace.jsonl";


function log(direction,data){

 let text=data.toString();


 let lines=text
 .split("\n")
 .filter(Boolean);


 for(const line of lines){

   try{

    const json=
     JSON.parse(line);


    fs.appendFileSync(
     logFile,
     JSON.stringify({

       time:
       new Date().toISOString(),

       direction,

       payload:json

     })+"\n"
    );


   }catch(e){

    fs.appendFileSync(
     logFile,
     JSON.stringify({

       time:
       new Date().toISOString(),

       direction,

       raw:line

     })+"\n"
    );

   }

 }

}


const child=
spawn(
 "codebuddy-code.exe",
 process.argv.slice(2),
 {
  stdio:[
   "pipe",
   "pipe",
   "inherit"
  ]
 });



process.stdin.on(
"data",
data=>{

 log(
  "paseo->agent",
  data
 );


 child.stdin.write(data);

});



child.stdout.on(
"data",
data=>{


 log(
  "agent->paseo",
  data
 );


 process.stdout.write(data);


});


child.on(
"exit",
code=>{

 process.exit(code);

});

```

                    Paseo

                ACP Protocol
                      |
                      |
                  cmd-acp
                      |
        +-------------+-------------+
        |                           |
        |                           |
 ToolCallStore              PermissionBroker
        |                           |
        |                           |
        |                    Named Pipe
        |                           |
        |                           |

这个架构的关键点：

**cmd-acp 不是给 command-code “加 ACP”，而是把 command-code 变成一个 headless agent，通过 ACP Client 获得 Human-in-the-loop 能力。**
